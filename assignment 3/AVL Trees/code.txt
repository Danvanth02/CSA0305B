#include <stdio.h>
#include <stdlib.h>

typedef struct N {
    int k;
    struct N *l, *r;
} N;

N *ins(N *t, int k) {
    if (!t) {
        t = (N *)calloc(1, sizeof *t);
        t->k = k;
        return t;
    }
    if (k < t->k) t->l = ins(t->l, k);
    else if (k > t->k) t->r = ins(t->r, k);
    return t;
}

N *del(N *t, int k) {
    if (!t) return NULL;
    if (k < t->k) t->l = del(t->l, k);
    else if (k > t->k) t->r = del(t->r, k);
    else if (t->l && t->r) {
        N *s = t->r;
        while (s->l) s = s->l; // Inorder successor
        t->k = s->k;
        t->r = del(t->r, s->k);
    } else {
        N *temp = t->l ? t->l : t->r;
        free(t);
        return temp;
    }
    return t;
}

void preorder(N *t) {
    if (!t) return;
    printf("%d ", t->k);
    preorder(t->l);
    preorder(t->r);
}

int main() {
    N *root = NULL;
    int keys[] = {50, 30, 70, 20, 40, 60, 80, 65};
    for (int i = 0; i < 8; i++) root = ins(root, keys[i]);

    // Deletions: leaf (20), one child (30), two children (50)
    root = del(root, 20);
    root = del(root, 30);
    root = del(root, 50);

    preorder(root);
    printf("\n");
    return 0;
}
